type Actor interface {
        Receive(ctx Context)
}

type Lifecycle interface {
        OnStart(ctx Context)
        OnStop(ctx Context)
}

type ActorRef struct {
        name    string
        mailbox Mailbox
        engine  *Engine
        done    chan struct{}
}

func (r *ActorRef) Name() string {
        return r.name
}

func (r *ActorRef) Done() <-chan struct{} {
        return r.done
}

func (r *ActorRef) Tell(msg any) error {
        if !r.mailbox.Push(Envelope{
                Sender:  nil,
                Message: msg,
        }) {
                return fmt.Errorf("actor %s is stopped", r.name)
        }
        return nil
}

func (r *ActorRef) TellFrom(sender *ActorRef, msg any) error {
        if !r.mailbox.Push(Envelope{
                Sender:  sender,
                Message: msg,
        }) {
                return fmt.Errorf("actor %s is stopped", r.name)
        }
        return nil
}

func (r *ActorRef) Ask(msg any, timeout time.Duration) (any, error) {
        f := newFuture()
        if !r.mailbox.Push(Envelope{
                Message: msg,
                replyTo: f,
        }) {
                return nil, fmt.Errorf("actor %s is stopped", r.name)
        }
        return f.wait(timeout)
}

// Context 提供 actor 处理消息时的上下文信息和操作能力
type Context interface {
        // Self 返回当前 actor 的引用
        Self() *ActorRef
        // Sender 返回发送当前消息的 actor 引用，如果是外部发送则返回 nil
        Sender() *ActorRef
        // Message 返回当前正在处理的消息
        Message() any
        // Reply 向消息发送者回复消息，仅在 Ask 模式下有效
        Reply(msg any) error
        // Spawn 创建并启动一个新的子 actor
        Spawn(name string, a Actor) (*ActorRef, error)
        // Stop 停止指定的 actor
        Stop(ref *ActorRef)
        // Watch 监听指定 actor 的终止事件，当该 actor 停止时会收到 Terminated 消息
        Watch(ref *ActorRef)
        // Unwatch 取消对指定 actor 的监听
        Unwatch(ref *ActorRef)
        // ScheduleAfter 在指定时间后向当前 actor 发送一条消息，返回可取消的定时器
        ScheduleAfter(d time.Duration, msg any) *Timer
}

type actorContext struct {
        self    *ActorRef
        engine  *Engine
        current *Envelope
        replied bool
}

func (c *actorContext) Self() *ActorRef {
        return c.self
}

func (c *actorContext) Sender() *ActorRef {
        if c.current == nil {
                return nil
        }
        return c.current.Sender
}

func (c *actorContext) Message() any {
        if c.current == nil {
                return nil
        }
        return c.current.Message
}

func (c *actorContext) Spawn(name string, a Actor) (*ActorRef, error) {
        return c.engine.Spawn(name, a)
}

func (c *actorContext) Stop(ref *ActorRef) {
        c.engine.Stop(ref.Name())
}

func (c *actorContext) Watch(ref *ActorRef) {
        if c.engine.watchManager.add(c.self, ref) {
                c.engine.metrics.WatchAdded(c.self.name, ref.name)
        }
}

func (c *actorContext) Unwatch(ref *ActorRef) {
        if c.engine.watchManager.remove(c.self, ref) {
                c.engine.metrics.WatchRemoved(c.self.name, ref.name)
        }
}

func (c *actorContext) ScheduleAfter(d time.Duration, msg any) *Timer {
        return &Timer{
                timer: time.AfterFunc(d, func() {
                        c.self.Tell(msg)
                }),
        }
}

func (c *actorContext) Reply(msg any) error {
        if c.current == nil || c.current.replyTo == nil {
                return fmt.Errorf("Reply called on a non-Ask message")
        }
        if c.replied {
                return fmt.Errorf("Reply already called for current message")
        }
        c.replied = true
        c.current.replyTo.reply(msg)
        return nil
}

func (c *actorContext) cleanup() {
}

type actorInstance struct {
        actor   Actor
        ref     *ActorRef
        ctx     *actorContext
        receive ReceiveFunc
}

func (inst *actorInstance) triggerOnStart() {
        if lifecycle, ok := inst.actor.(Lifecycle); ok {
                lifecycle.OnStart(inst.ctx)
        }
}

func (inst *actorInstance) triggerOnStop() {
        if lifecycle, ok := inst.actor.(Lifecycle); ok {
                lifecycle.OnStop(inst.ctx)
        }
}

func (inst *actorInstance) dispatch(env Envelope, onPanic func(any)) {
        defer func() {
                if r := recover(); r != nil {
                        onPanic(r)
                }
        }()

        inst.ctx.current = &env
        inst.ctx.replied = false
        inst.receive(inst.ctx)
        inst.ctx.current = nil
}

type Engine struct {
        actors         map[string]*actorInstance
        mu             sync.RWMutex
        logger         Logger
        metrics        Metrics
        middleware     []Middleware
        mailboxFactory MailboxFactory
        watchManager   *watchManager
}

var ErrActorExists = errors.New("actor already exists")

func NewEngine(opts ...EngineOption) *Engine {
        e := &Engine{
                actors:         make(map[string]*actorInstance),
                logger:         &defaultLogger{},
                metrics:        noopMetrics{},
                mailboxFactory: func() Mailbox { return NewChanMailbox(256) },
                watchManager:   newWatchManager(),
        }
        for _, opt := range opts {
                opt(e)
        }
        return e
}

func (e *Engine) Metrics() Metrics {
        return e.metrics
}

func (e *Engine) Spawn(name string, a Actor) (*ActorRef, error) {
        e.mu.Lock()
        defer e.mu.Unlock()

        if _, exists := e.actors[name]; exists {
                return nil, ErrActorExists
        }

        inst := e.newActorInstance(name, a)
        e.actors[name] = inst
        e.metrics.ActorSpawned(name)
        go e.processLoop(inst)

        return inst.ref, nil
}

func (e *Engine) newActorInstance(name string, a Actor) *actorInstance {
        mailbox := e.mailboxFactory()
        ref := &ActorRef{
                name:    name,
                mailbox: mailbox,
                engine:  e,
                done:    make(chan struct{}),
        }

        receive := ReceiveFunc(a.Receive)
        for i := len(e.middleware) - 1; i >= 0; i-- {
                receive = e.middleware[i](receive)
        }

        return &actorInstance{
                actor:   a,
                ref:     ref,
                ctx:     &actorContext{self: ref, engine: e},
                receive: receive,
        }
}

func (e *Engine) processLoop(inst *actorInstance) {
        defer func() {
                inst.ctx.cleanup()
                e.watchManager.cleanupForWatcher(inst.ref)
                e.metrics.ActorStopped(inst.ref.name)
                close(inst.ref.done)
        }()

        inst.triggerOnStart()

        for {
                env, ok := inst.ref.mailbox.Pull()
                if !ok {
                        inst.triggerOnStop()
                        return
                }

                if _, ok := env.Message.(PoisonPill); ok {
                        inst.triggerOnStop()
                        return
                }

                start := time.Now()
                inst.dispatch(env, func(r any) {
                        e.logger.Printf("[actor] %s panic: %v", inst.ref.name, r)
                        e.metrics.PanicRecovered(inst.ref.name)
                })
                e.metrics.MessageProcessed(inst.ref.name, time.Since(start))
        }
}

func (e *Engine) Lookup(name string) (*ActorRef, bool) {
        e.mu.RLock()
        defer e.mu.RUnlock()

        inst, ok := e.actors[name]
        if !ok {
                return nil, false
        }
        return inst.ref, true
}

func (e *Engine) Stop(name string) {
        e.mu.Lock()
        inst, ok := e.actors[name]
        if !ok {
                e.mu.Unlock()
                return
        }
        delete(e.actors, name)
        e.mu.Unlock()

        inst.ref.mailbox.Stop()
}

func (e *Engine) Shutdown(ctx ...context.Context) error {
        e.mu.Lock()
        instances := make([]*actorInstance, 0, len(e.actors))
        for name, inst := range e.actors {
                instances = append(instances, inst)
                delete(e.actors, name)
        }
        e.mu.Unlock()

        for _, inst := range instances {
                inst.ref.mailbox.Stop()
        }

        var deadline <-chan struct{}
        if len(ctx) > 0 && ctx[0] != nil {
                deadline = ctx[0].Done()
        }

        for _, inst := range instances {
                select {
                case <-inst.ref.done:
                case <-deadline:
                        return fmt.Errorf("engine shutdown timed out, %d actors still running", e.runningCount(instances))
                }
        }
        return nil
}

func (e *Engine) runningCount(instances []*actorInstance) int {
        count := 0
        for _, inst := range instances {
                select {
                case <-inst.ref.done:
                default:
                        count++
                }
        }
        return count
}

type Envelope struct {
        Sender  *ActorRef
        Message any
        replyTo *future
}

type Mailbox interface {
        Push(e Envelope) bool
        Pull() (Envelope, bool)
        Stop()
}

type ChanMailbox struct {
        ch   chan Envelope
        done chan struct{}
        once sync.Once
}

func NewChanMailbox(size int) *ChanMailbox {
        return &ChanMailbox{
                ch:   make(chan Envelope, size),
                done: make(chan struct{}),
        }
}

func (m *ChanMailbox) Push(e Envelope) bool {
        select {
        case <-m.done:
                return false
        default:
        }
        select {
        case <-m.done:
                return false
        case m.ch <- e:
                return true
        }
}

func (m *ChanMailbox) Pull() (Envelope, bool) {
        select {
        case env, ok := <-m.ch:
                return env, ok
        case <-m.done:
                select {
                case env := <-m.ch:
                        return env, true
                default:
                        return Envelope{}, false
                }
        }
}

func (m *ChanMailbox) Stop() {
        m.once.Do(func() {
                close(m.done)
        })
}

type future struct {
        ch chan any
}

func newFuture() *future {
        return &future{ch: make(chan any, 1)}
}

func (f *future) reply(msg any) bool {
        select {
        case f.ch <- msg:
                return true
        default:
                return false
        }
}

func (f *future) wait(timeout time.Duration) (any, error) {
        if timeout < 0 {
                return <-f.ch, nil
        }
        if timeout == 0 {
                select {
                case reply := <-f.ch:
                        return reply, nil
                default:
                        return nil, fmt.Errorf("ask timeout: immediate mode")
                }
        }
        ctx, cancel := context.WithTimeout(context.Background(), timeout)
        defer cancel()
        select {
        case reply := <-f.ch:
                return reply, nil
        case <-ctx.Done():
                return nil, fmt.Errorf("ask timeout after %v", timeout)
        }
}

type Metrics interface {
        ActorSpawned(name string)
        ActorStopped(name string)
        MessageProcessed(name string, duration time.Duration)
        MessageQueued(name string)
        MessageDropped(name string)
        PanicRecovered(name string)
        WatchAdded(watcher, watched string)
        WatchRemoved(watcher, watched string)
        GetSnapshot() MetricsSnapshot
}

type MetricsSnapshot struct {
        TotalActors   int64
        TotalMessages int64
        TotalPanics   int64
        TotalWatches  int64
        ActorStats    map[string]ActorStats
}

type ActorStats struct {
        Name              string
        MessagesQueued    int64
        MessagesProcessed int64
        Panics            int64
        ActiveWatches     int64
}

type defaultMetrics struct {
        totalActors   atomic.Int64
        totalMessages atomic.Int64
        totalPanics   atomic.Int64
        totalWatches  atomic.Int64

        mu         sync.RWMutex
        actorStats map[string]*actorStatsEntry
}

type actorStatsEntry struct {
        messagesQueued    atomic.Int64
        messagesProcessed atomic.Int64
        panics            atomic.Int64
        activeWatches     atomic.Int64
}

func NewDefaultMetrics() Metrics {
        return &defaultMetrics{
                actorStats: make(map[string]*actorStatsEntry),
        }
}

func (m *defaultMetrics) getOrCreateEntry(name string) *actorStatsEntry {
        m.mu.RLock()
        entry, ok := m.actorStats[name]
        m.mu.RUnlock()
        if ok {
                return entry
        }

        m.mu.Lock()
        defer m.mu.Unlock()
        entry, ok = m.actorStats[name]
        if ok {
                return entry
        }
        entry = &actorStatsEntry{}
        m.actorStats[name] = entry
        return entry
}

func (m *defaultMetrics) ActorSpawned(name string) {
        m.totalActors.Add(1)
        m.getOrCreateEntry(name)
}

func (m *defaultMetrics) ActorStopped(name string) {
        m.totalActors.Add(-1)
}

func (m *defaultMetrics) MessageProcessed(name string, duration time.Duration) {
        m.totalMessages.Add(1)
        entry := m.getOrCreateEntry(name)
        entry.messagesProcessed.Add(1)
}

func (m *defaultMetrics) MessageQueued(name string) {
        entry := m.getOrCreateEntry(name)
        entry.messagesQueued.Add(1)
}

func (m *defaultMetrics) MessageDropped(name string) {
        entry := m.getOrCreateEntry(name)
        entry.messagesQueued.Add(-1)
}

func (m *defaultMetrics) PanicRecovered(name string) {
        m.totalPanics.Add(1)
        entry := m.getOrCreateEntry(name)
        entry.panics.Add(1)
}

func (m *defaultMetrics) WatchAdded(watcher, watched string) {
        m.totalWatches.Add(1)
        entry := m.getOrCreateEntry(watcher)
        entry.activeWatches.Add(1)
}

func (m *defaultMetrics) WatchRemoved(watcher, watched string) {
        m.totalWatches.Add(-1)
        entry := m.getOrCreateEntry(watcher)
        entry.activeWatches.Add(-1)
}

func (m *defaultMetrics) GetSnapshot() MetricsSnapshot {
        m.mu.RLock()
        defer m.mu.RUnlock()

        actorStats := make(map[string]ActorStats, len(m.actorStats))
        for name, entry := range m.actorStats {
                actorStats[name] = ActorStats{
                        Name:              name,
                        MessagesQueued:    entry.messagesQueued.Load(),
                        MessagesProcessed: entry.messagesProcessed.Load(),
                        Panics:            entry.panics.Load(),
                        ActiveWatches:     entry.activeWatches.Load(),
                }
        }

        return MetricsSnapshot{
                TotalActors:   m.totalActors.Load(),
                TotalMessages: m.totalMessages.Load(),
                TotalPanics:   m.totalPanics.Load(),
                TotalWatches:  m.totalWatches.Load(),
                ActorStats:    actorStats,
        }
}

type noopMetrics struct{}

func (noopMetrics) ActorSpawned(name string)                             {}
func (noopMetrics) ActorStopped(name string)                             {}
func (noopMetrics) MessageProcessed(name string, duration time.Duration) {}
func (noopMetrics) MessageQueued(name string)                            {}
func (noopMetrics) MessageDropped(name string)                           {}
func (noopMetrics) PanicRecovered(name string)                           {}
func (noopMetrics) WatchAdded(watcher, watched string)                   {}
func (noopMetrics) WatchRemoved(watcher, watched string)                 {}
func (noopMetrics) GetSnapshot() MetricsSnapshot {
        return MetricsSnapshot{ActorStats: make(map[string]ActorStats)}
}

package actor

import "sync"

type watchedState struct {
        watchers map[*ActorRef]struct{}
        stopCh   chan struct{}
}

type watchManager struct {
        mu      sync.RWMutex
        watches map[*ActorRef]*watchedState
}

func newWatchManager() *watchManager {
        return &watchManager{
                watches: make(map[*ActorRef]*watchedState),
        }
}

func (wm *watchManager) add(watcher, watched *ActorRef) bool {
        select {
        case <-watched.Done():
                watcher.Tell(Terminated{Who: watched})
                return false
        default:
        }

        wm.mu.Lock()
        defer wm.mu.Unlock()

        state, ok := wm.watches[watched]
        if !ok {
                state = &watchedState{
                        watchers: make(map[*ActorRef]struct{}),
                        stopCh:   make(chan struct{}),
                }
                wm.watches[watched] = state
                go wm.monitor(watched, state.stopCh)
        }

        if _, exists := state.watchers[watcher]; exists {
                return false
        }

        state.watchers[watcher] = struct{}{}
        return true
}

func (wm *watchManager) remove(watcher, watched *ActorRef) bool {
        wm.mu.Lock()
        defer wm.mu.Unlock()

        state, ok := wm.watches[watched]
        if !ok {
                return false
        }

        if _, exists := state.watchers[watcher]; !exists {
                return false
        }

        delete(state.watchers, watcher)

        if len(state.watchers) == 0 {
                close(state.stopCh)
                delete(wm.watches, watched)
        }

        return true
}

func (wm *watchManager) cleanupForWatcher(watcher *ActorRef) {
        wm.mu.Lock()
        defer wm.mu.Unlock()

        var toRemove []*ActorRef
        var toClose []chan struct{}
        for watched, state := range wm.watches {
                if _, exists := state.watchers[watcher]; exists {
                        delete(state.watchers, watcher)
                        if len(state.watchers) == 0 {
                                toRemove = append(toRemove, watched)
                                toClose = append(toClose, state.stopCh)
                        }
                }
        }
        for i, watched := range toRemove {
                close(toClose[i])
                delete(wm.watches, watched)
        }
}

func (wm *watchManager) monitor(watched *ActorRef, stopCh <-chan struct{}) {
        select {
        case <-watched.Done():
        case <-stopCh:
                return
        }

        wm.mu.Lock()
        state := wm.watches[watched]
        watchers := make([]*ActorRef, 0, len(state.watchers))
        for watcher := range state.watchers {
                watchers = append(watchers, watcher)
        }
        delete(wm.watches, watched)
        wm.mu.Unlock()

        for _, watcher := range watchers {
                watcher.Tell(Terminated{Who: watched})
        }
}
